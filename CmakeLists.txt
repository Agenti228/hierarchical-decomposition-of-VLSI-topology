cmake_minimum_required(VERSION 3.13)

# =========================================
# GLOBAL SETTINGS
# =========================================

# Только static библиотеки
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

# Полностью static runtime MSVC (/MT)
set(CMAKE_MSVC_RUNTIME_LIBRARY
    "MultiThreaded$<$<CONFIG:Debug>:Debug>"
    CACHE STRING "" FORCE)

project(vlsi_topology)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# =========================================
# ZLIB
# =========================================

set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)

add_subdirectory(thirdParty/zlib)

# alias target
add_library(ZLIB::ZLIB ALIAS zlibstatic)

# variables for find_package(ZLIB)
set(ZLIB_FOUND TRUE CACHE BOOL "" FORCE)

set(ZLIB_LIBRARY
    zlibstatic
    CACHE STRING "" FORCE)

set(ZLIB_LIBRARIES
    zlibstatic
    CACHE STRING "" FORCE)

set(ZLIB_INCLUDE_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/zlib"
    CACHE PATH "" FORCE)

set(ZLIB_INCLUDE_DIRS
    "${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/zlib"
    CACHE PATH "" FORCE)

# =========================================
# QHULL
# =========================================

set(QHULL_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)

add_subdirectory(thirdParty/qhull)

# alias target
add_library(Qhull::qhullstatic_r ALIAS qhullstatic_r)

# variables for find_package(Qhull)
set(QHULL_FOUND TRUE CACHE BOOL "" FORCE)
set(Qhull_FOUND TRUE CACHE BOOL "" FORCE)

set(Qhull_LIBRARY
    qhullstatic_r
    CACHE STRING "" FORCE)

set(Qhull_LIBRARIES
    qhullstatic_r
    CACHE STRING "" FORCE)

set(Qhull_INCLUDE_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/qhull/src"
    CACHE PATH "" FORCE)

set(Qhull_INCLUDE_DIRS
    "${CMAKE_CURRENT_SOURCE_DIR}/thirdParty/qhull/src"
    CACHE STRING "" FORCE)

# =========================================
# GDSTK
# =========================================

set(GDSTK_PYTHON OFF CACHE BOOL "" FORCE)

add_subdirectory(thirdParty/gdstk)

# =========================================
# APP
# =========================================

file(GLOB_RECURSE MY_SOURCES
    CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)

add_executable(${PROJECT_NAME} ${MY_SOURCES})

target_compile_options(${PROJECT_NAME}
    PRIVATE
    /utf-8
)

target_include_directories(${PROJECT_NAME}
    PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
)

target_link_libraries(${PROJECT_NAME}
    PRIVATE
    gdstk
    zlibstatic
    qhullstatic_r
)

# =========================================
# WINDOWS STATIC DEFINES
# =========================================

target_compile_definitions(${PROJECT_NAME}
    PRIVATE
    _CRT_SECURE_NO_WARNINGS
)

# =========================================
# OUTPUT FOLDERS
# =========================================

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

foreach(CONFIG Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER ${CONFIG} CONFIG_UPPER)

    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER}
        ${CMAKE_BINARY_DIR}/bin/${CONFIG})

    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${CONFIG_UPPER}
        ${CMAKE_BINARY_DIR}/bin/${CONFIG})

    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${CONFIG_UPPER}
        ${CMAKE_BINARY_DIR}/lib/${CONFIG})
endforeach()
